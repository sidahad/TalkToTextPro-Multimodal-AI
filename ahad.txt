# ai_engine_gemini.py
import os, base64
from dotenv import load_dotenv
import google.generativeai as genai

load_dotenv()
genai.configure(api_key=os.getenv("GEMINI_API_KEY"))

# create model
model = genai.GenerativeModel("models/gemini-1.5-flash")

# start a chat session (keeps history inside Gemini session)
chat_session = model.start_chat(history=[])

def get_answer(query: str) -> str:
    try:
        response = chat_session.send_message(query)
        return response.text
    except Exception as e:
        return f"Error generating answer: {str(e)}"

def get_answer_with_image(query: str, image_file):
    try:
        if not image_file:
            return get_answer(query), None
        # read bytes
        img_bytes = image_file.read()
        # send image + text to model
        # note: mime_type may be auto-detected; using image/png as generic
        response = model.generate_content([
            {"mime_type": "image/png", "data": img_bytes},
            query
        ])
        encoded_image = base64.b64encode(img_bytes).decode("utf-8")
        return response.text, encoded_image
    except Exception as e:
        return f"Error: {str(e)}", None

# pdf_processing.py
import fitz  # PyMuPDF

def extract_text_from_pdf(file):
    """
    file: Werkzeug FileStorage from Flask (file.read() available)
    returns extracted text or an error message starting with 'Error'
    """
    try:
        data = file.read()
        if not data:
            return "No data in PDF file."
        doc = fitz.open(stream=data, filetype="pdf")
        text = ""
        for page in doc:
            text += page.get_text()
        doc.close()
        return text if text.strip() else "No extractable text found in PDF."
    except Exception as e:
        return f"Error extracting PDF text: {str(e)}"

        return jsonify({"error": "Empty query"}), 400

@app.route("/download_notes", methods=["GET"])
def download_notes():
    global last_notes
    if not last_notes:
        return "No notes available", 400
    tmp = tempfile.NamedTemporaryFile(delete=False, suffix=".txt")
    tmp.write(last_notes.encode("utf-8"))
    tmp.flush()
    tmp.close()
    return send_file(tmp.name, as_attachment=True, download_name="meeting_notes.txt")

    const voices = synth.getVoices();

function speakText(text) {
    if (isSpeaking || !synth) return;
    isSpeaking = true;
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.voice = loadVoices();
    utterance.lang = 'en-US';
    utterance.onend = () => {
        isSpeaking = false;
    };
    synth.speak(utterance);
}

function saveChatLocal(chatId, sender, text, image = null, fileName = null) {
    let chats = JSON.parse(localStorage.getItem("chats") || "{}");
    if (!chats[chatId]) chats[chatId] = [];
    chats[chatId].push({ sender, text, image, fileName });
    localStorage.setItem("chats", JSON.stringify(chats));
    renderChatHistory();
}

function loadChat(chatId) {
    if (chatContainer) {
        chatContainer.innerHTML = "";
    }
    currentChat = chatId;
    let chats = JSON.parse(localStorage.getItem("chats") || "{}");
    (chats[chatId] || []).forEach(m => addMessage(m.text, m.sender, m.image, m.fileName));
}

function deleteChat(chatId) {
    let chats = JSON.parse(localStorage.getItem("chats") || "{}");
    delete chats[chatId];
    localStorage.setItem("chats", JSON.stringify(chats));
    let pinnedList = JSON.parse(localStorage.getItem("pinned") || "[]");
    pinnedList = pinnedList.filter(id => id !== chatId);
    localStorage.setItem("pinned", JSON.stringify(pinnedList));
    renderChatHistory();
    if (currentChat === chatId) {
        currentChat = "chat_" + Date.now();
        if (chatContainer) {
            chatContainer.innerHTML = "";
        }
        saveChatLocal(currentChat, "system", "New chat started.");
    }
}

function renderChatHistory() {
    const chatHistoryDiv = document.getElementById("chat-history");
    const pinnedDiv = document.getElementById("pinned-history");

    pinnedDiv.innerHTML = "";
    let chats = JSON.parse(localStorage.getItem("chats") || "{}");
    let pinned = JSON.parse(localStorage.getItem("pinned") || "[]");

    Object.keys(chats).forEach(id => {
        const wrapper = document.createElement("div");
        wrapper.style.display = "flex";
        wrapper.style.alignItems = "center";
        const btn = document.createElement("button");
        btn.textContent = id;
        btn.className = "load-btn";
        btn.onclick = () => loadChat(id);
        const pinBtn = document.createElement("button");
        pinBtn.textContent = pinned.includes(id) ? "📌" : "📍";
        pinBtn.onclick = (e) => { e.stopPropagation(); togglePin(id); };
        const deleteBtn = document.createElement("button");
        deleteBtn.textContent = "🗑️";
        deleteBtn.style.marginLeft = "5px";
        deleteBtn.onclick = (e) => { e.stopPropagation(); if (confirm("Delete this chat?")) deleteChat(id); };

        wrapper.appendChild(btn);
        wrapper.appendChild(pinBtn);
        wrapper.appendChild(deleteBtn);

        if (pinned.includes(id)) pinnedDiv.appendChild(wrapper); else chatHistoryDiv.appendChild(wrapper);
    });
    filterHistory();

function togglePin(id) {
    let pinned = JSON.parse(localStorage.getItem("pinned") || "[]");
    if (pinned.includes(id)) pinned = pinned.filter(x => x !== id); else pinned.push(id);
    localStorage.setItem("pinned", JSON.stringify(pinned));
    renderChatHistory();
}

});

function filterHistory() {
    const term = searchInput.value.toLowerCase();
    const wrappers = document.querySelectorAll('.chat-history > div');
    wrappers.forEach(wrapper => {
        const btn = wrapper.querySelector('.load-btn');
        if (btn.textContent.toLowerCase().includes(term)) {
            wrapper.style.display = 'flex';
        } else {
            wrapper.style.display = 'none';
        }
    });
}

This is my audio_processing.py
# audio_processing.py
import speech_recognition as sr
from pydub import AudioSegment
from deep_translator import GoogleTranslator
import tempfile
import os

def audio_file_to_text(file_storage, lang="en"):
    """
    file_storage: Werkzeug FileStorage object (from Flask request.files['audio'])
    lang: language code used by Google recognizer (e.g., 'en', 'ur', 'fr')
    returns: transcript text
    """
    # save to temp file (preserve original extension if available)
    suffix = ""
    if hasattr(file_storage, "filename") and "." in file_storage.filename:
        suffix = os.path.splitext(file_storage.filename)[1]
    tmp = tempfile.NamedTemporaryFile(delete=False, suffix=suffix or ".mp3")
    file_storage.save(tmp.name)
    tmp.flush()
    tmp.close()

    # convert to wav (SpeechRecognition works well with WAV)
    wav_tmp = tempfile.NamedTemporaryFile(delete=False, suffix=".wav")
    sound = AudioSegment.from_file(tmp.name)
    sound.export(wav_tmp.name, format="wav")
    wav_tmp.flush()
    wav_tmp.close()

    try:
        r = sr.Recognizer()
        with sr.AudioFile(wav_tmp.name) as source:
            audio = r.record(source)
            text = r.recognize_google(audio, language=lang)
    finally:
        # cleanup temp files
        try:
            os.remove(tmp.name)
        except: pass
        try:
            os.remove(wav_tmp.name)
        except: pass

    return text

def translate_to_english(text, src_lang="en"):
    """
    Translate 'text' to English if src_lang != 'en'.
    If translation fails, returns original text.
    """
    try:
        if not text.strip():
            return text
        if src_lang.lower() == "en":
            return text
        return GoogleTranslator(source=src_lang, target="en").translate(text)
    except Exception:
        return text

import sqlite3
from pymongo import MongoClient
import os
from dotenv import load_dotenv
import bcrypt

load_dotenv()
MONGO_URI = os.getenv("MONGO_URI", "mongodb://localhost:27017/")
client = MongoClient(MONGO_URI)
db_mongo = client["talktotext"]
notes_collection = db_mongo["meeting_notes"]

# SQLite database setup
SQLITE_DB = "users.db"

def init_sqlite_db():
    conn = sqlite3.connect(SQLITE_DB)
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password TEXT NOT NULL
        )
    """)
    conn.commit()
    conn.close()

# Initialize the database on import
init_sqlite_db()

def save_user(username, password):
    """
    Save a new user to SQLite database with hashed password.
    Returns True if successful, False if username already exists.
    """
    try:
        conn = sqlite3.connect(SQLITE_DB)
        cursor = conn.cursor()
        hashed_password = bcrypt.hashpw(password.encode('utf-8'), bcrypt.gensalt())
        cursor.execute("INSERT INTO users (username, password) VALUES (?, ?)", (username, hashed_password))
        conn.commit()
        conn.close()
        return True
    except sqlite3.IntegrityError:
        conn.close()
        return False

def check_user(username, password):
    """
    Check if user exists and password is correct.
    Returns True if credentials are valid, False otherwise.
    """
    conn = sqlite3.connect(SQLITE_DB)
    cursor = conn.cursor()
    cursor.execute("SELECT password FROM users WHERE username = ?", (username,))
    result = cursor.fetchone()
    conn.close()
    if result:
        return bcrypt.checkpw(password.encode('utf-8'), result[0])
    return False

def save_notes(user, raw, translated, cleaned, final_notes):
    notes_collection.insert_one({
        "user": user,
        "raw_transcript": raw,
        "translated": translated,
        "cleaned": cleaned,
        "final_notes": final_notes
    })

def get_notes(user):
    return list(notes_collection.find({"user": user}))
